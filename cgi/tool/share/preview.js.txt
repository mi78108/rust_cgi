/* preview.js.txt — 分享预览（页面级公共逻辑，page.html 与 public.html 共用）
 *
 * 实际以 /tool/share/preview.js 提供（由 tool/share/index 读本文件吐出去）。
 * 文件名带 .txt 是刻意的 —— spider 会把 URL 上真实存在的文件当 CGI 脚本
 * 去 exec，叫 preview.js 会被 Ruby 执行而崩。见 index 里「共享静态资源」注释。
 *
 * 用法：<script src="/tool/share/preview.js"></script>
 *   previewCfg({ box: '#my-list' })   // 声明列表容器（选择器）
 *   previewKind(item)                 // 'image' | 'audio' | 'video' | 'text' | null
 *   togglePreview(item, el)           // 展开/收起
 *   closeAllPreviews()
 *
 * 为什么独立成文件：预览逻辑（类型判定 + 四种渲染 + 展开收起）有 ~100 行，
 * 在两个页面各抄一遍必然走样。共享一份，两个页面只各自管列表数据。
 *
 * 预览走 /tool/share/preview/<token> 而不是 /download/f：
 *   前者 inline 渲染且不扣下载次数，服务端只放行 pub=1；
 *   后者 attachment + 验密码 + 每次扣次数。
 */

const TEXT_EXT = /\.(txt|log|md|markdown|json|jsonl|xml|ya?ml|html?|css|js|mjs|cjs|ts|tsx|jsx|py|rb|go|rs|c|h|cpp|hpp|java|kt|kts|swift|scala|sh|bash|zsh|fish|sql|ini|cfg|conf|toml|env|gitignore|dockerfile|makefile|csv|tsv|properties|gradle|tf|tfvars|vue|svelte|less|scss|sass|patch|diff|log\.\d+|txt\.log)$/i;
const IMG_EXT  = /\.(png|jpe?g|gif|webp|bmp|avif|svg|ico)$/i;

const PREVIEW_MAX = 512 * 1024;      // 文本预览上限 512KB —— 文本要整个读进内存再贴到 <pre>，太大没意义
const IMAGE_MAX   = 16 * 1024 * 1024; // 图片内嵌 <img>，别太大

// 列表容器选择器，由各页面声明（page.html 是 #my-list，public.html 是 #list）
let PREVIEW_BOX = '#my-list';
function previewCfg(o) { if (o && o.box) PREVIEW_BOX = o.box; }

// 能不能预览？能的话是哪种。
// 私密（pub=0）不给：preview 端点只服务 pub=1，而且私密要先过密码才有意义。
function previewKind(it) {
  if (it.pub === 0) return null;
  const t = (it.type || '').toLowerCase();
  const n = it.name || '';
  if (/^image\//.test(t) || IMG_EXT.test(n)) return it.size <= IMAGE_MAX ? 'image' : null;
  // 音视频不设上限：preview 端点支持 Range（206），浏览器只会拉它需要的片段，
  // 几百 MB 的视频照样能预览/拖动 —— 早期那版 64MB 上限纯属多余，
  // 反而把最大的视频挡在门外（752MB 的 mp4 就没有预览按钮）。
  if (/^audio\//.test(t)) return 'audio';
  if (/^video\//.test(t)) return 'video';
  const texty = /^text\//.test(t) || /json|xml|javascript|yaml|x-sh/.test(t) || TEXT_EXT.test(n);
  if (texty) return it.size <= PREVIEW_MAX ? 'text' : null;
  return null;
}

function pvUrl(it) { return `/tool/share/preview/${it.token}`; }

function errNode(msg) {
  const d = document.createElement('div');
  d.className = 'pv-err';
  d.textContent = msg;
  return d;
}

// 收起所有预览（停止播放、释放资源），同一时刻只展开一个
function closeAllPreviews() {
  document.querySelectorAll('.pv').forEach((pv) => {
    const v = pv.querySelector('video, audio');
    if (v) { try { v.pause(); v.removeAttribute('src'); v.load(); } catch (_) {} }
    if (pv.dataset.objurl) URL.revokeObjectURL(pv.dataset.objurl);
    pv.remove();
  });
  document.querySelectorAll('.item.open').forEach((it) => it.classList.remove('open'));
}

// 展开/收起某个分享的预览。el 是它所在的 .item 容器。
function togglePreview(it, el) {
  if (el.querySelector('.pv')) { closeAllPreviews(); return; }
  closeAllPreviews();
  el.classList.add('open');

  const kind = previewKind(it);
  const pv = document.createElement('div');
  pv.className = 'pv';
  pv.innerHTML =
    `<div class="pv-head"><span>preview</span><span class="spacer"></span>` +
    `<button class="pv-close" type="button">close</button></div>`;
  pv.querySelector('.pv-close').addEventListener('click', closeAllPreviews);
  el.querySelector('.meta').insertAdjacentElement('afterend', pv);

  const url = pvUrl(it);

  if (kind === 'image') {
    const img = document.createElement('img');
    img.src = url; img.alt = it.name || 'preview'; img.loading = 'lazy';
    img.onerror = () => { img.replaceWith(errNode('图片加载失败')); };
    pv.appendChild(img);
    return;
  }
  if (kind === 'audio') {
    const a = document.createElement('audio');
    a.controls = true; a.preload = 'metadata'; a.src = url;
    a.addEventListener('error', () => pv.appendChild(errNode('音频加载失败')));
    pv.appendChild(a);
    return;
  }
  if (kind === 'video') {
    const v = document.createElement('video');
    v.controls = true; v.preload = 'metadata'; v.playsInline = true; v.src = url;
    v.addEventListener('error', () => pv.appendChild(errNode('视频加载失败（可能编码不支持）')));
    pv.appendChild(v);
    return;
  }

  // 纯文本：取回来贴到 <pre>
  const note = errNode('加载中…');
  pv.appendChild(note);
  fetch(url)
    .then(async (res) => {
      if (!res.ok) {
        let msg = 'HTTP ' + res.status;
        try { const j = await res.json(); if (j && j.error) msg = j.error; } catch (_) {}
        throw new Error(msg);
      }
      return res.text();
    })
    .then((txt) => {
      pv.removeChild(note);
      const pre = document.createElement('pre');
      pre.textContent = txt;   // textContent，不是 innerHTML —— 防止文件内容当 HTML 执行
      pv.appendChild(pre);
    })
    .catch((e) => { note.textContent = '预览失败: ' + e.message; });
}
